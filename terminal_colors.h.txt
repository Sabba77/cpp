/**
 * @file terminal_colors.h
 * @brief Raccolta completa e documentata dei codici di escape ANSI per C e C++.
 * @author Sviluppato come prontuario completo per la personalizzazione di interfacce CLI.
 * * ====================================================================================
 * MANUALE RAPIDO D'USO E BEST PRACTICES:
 * ====================================================================================
 * 1. COSTRUTTO DEL CODICE: Ogni macro rappresenta una stringa di controllo. Il terminale
 * intercetta la sequenza "\x1b[" (dove \x1b è il carattere ESC in esadecimale), 
 * legge i parametri numerici separati da punti e virgola, e termina sul carattere 'm'.
 * * 2. EFFETTO PERSISTENTE: Quando invii un codice colore o uno stile, questo rimane attivo 
 * per TUTTI i successivi output (anche dopo la chiusura del programma, sporcando il 
 * prompt dell'utente). È FONDAMENTALE terminare sempre le stringhe con TRM_RESET.
 * * 3. CONCATENAZIONE AUTOMATICA: Il compilatore C/C++ unisce automaticamente le stringhe 
 * costanti adiacenti. Puoi formattare i tuoi printf senza usare specificatori %s.
 * Esempio: printf(TRM_B_RED TRM_BG_GRAY " ERRORE " TRM_RESET " File non trovato.\n");
 * * 4. COMPATIBILITÀ: Funziona nativamente su Linux, macOS e sui terminali Windows moderni 
 * (Windows Terminal / PowerShell aggiornata).
 * ====================================================================================
 */

#ifndef TERMINAL_COLORS_H
#define TERMINAL_COLORS_H

/* ====================================================================================
 * 1. STILI DI TESTO E MODIFICATORI STRUTTURALI
 * ====================================================================================
 * Questi codici alterano l'aspetto del carattere (font, peso, visibilità).
 */

#define TRM_RESET         "\x1b[0m"   /**< Ripristina lo stato predefinito del terminale (Rimuove stili e colori) */
#define TRM_BOLD          "\x1b[1m"   /**< Grassetto o Alta Intensità (Rende il testo più spesso o luminoso) */
#define TRM_FAINT         "\x1b[2m"   /**< Intensità dimezzata (Testo più opaco, utile per dettagli secondari) */
#define TRM_ITALIC        "\x1b[3m"   /**< Corsivo (Nota: Non tutti i terminali lo supportano, alcuni lo ignorano) */
#define TRM_UNDERLINE     "\x1b[4m"   /**< Sottolineato classico */
#define TRM_BLINK_SLOW    "\x1b[5m"   /**< Lampeggiante lento (Usare con estrema parsimonia, fastidioso nelle UI) */
#define TRM_REVERSE       "\x1b[7m"   /**< Invertito: Scambia il colore del testo con quello dello sfondo attuale */
#define TRM_CONCEAL       "\x1b[8m"   /**< INVISIBILE: Nasconde il testo digitato (Ideale per mascherare password o input sensibili) */
#define TRM_STRIKE        "\x1b[9m"   /**< Barrato (Traccia una linea orizzontale sul testo) */


/* ====================================================================================
 * 2. COLORI DEL TESTO (FOREGROUND) - PALETTE STANDARD (8 COLORI)
 * ====================================================================================
 * Colori standard adatti a qualsiasi terminale legacy o moderno.
 */

#define TRM_TXT_BLACK     "\x1b[30m"  /**< Testo Nero */
#define TRM_TXT_RED       "\x1b[31m"  /**< Testo Rosso (Ideale per errori o stati critici) */
#define TRM_TXT_GREEN     "\x1b[32m"  /**< Testo Verde (Ideale per messaggi di successo o conferme) */
#define TRM_TXT_YELLOW    "\x1b[33m"  /**< Testo Giallo (Ideale per avvisi, warning o attenzione) */
#define TRM_TXT_BLUE      "\x1b[34m"  /**< Testo Blu (Ottimo per intestazioni o categorie) */
#define TRM_TXT_MAGENTA   "\x1b[35m"  /**< Testo Magenta / Viola */
#define TRM_TXT_CYAN      "\x1b[36m"  /**< Testo Cyan / Azzurro (Molto leggibile su sfondi scuri) */
#define TRM_TXT_WHITE     "\x1b[37m"  /**< Testo Bianco */


/* ====================================================================================
 * 3. COLORI DEL TESTO (FOREGROUND) - ALTA INTENSITÀ / LUMINOSI (8 COLORI)
 * ====================================================================================
 * Versioni ad alta luminosità della palette base. Risaltano molto di più all'occhio.
 */

#define TRM_TXT_B_BLACK   "\x1b[90m"  /**< Grigio Scuro (Perfetto per commenti o timestamp nei log) */
#define TRM_TXT_B_RED     "\x1b[91m"  /**< Rosso Luminoso */
#define TRM_TXT_B_GREEN   "\x1b[92m"  /**< Verde Luminoso */
#define TRM_TXT_B_YELLOW  "\x1b[93m"  /**< Giallo Luminoso */
#define TRM_TXT_B_BLUE    "\x1b[94m"  /**< Blu Luminoso */
#define TRM_TXT_B_MAGENTA "\x1b[95m"  /**< Magenta Luminoso */
#define TRM_TXT_B_CYAN    "\x1b[96m"  /**< Cyan Luminoso */
#define TRM_TXT_B_WHITE   "\x1b[97m"  /**< Bianco Candido / Brillante */


/* ====================================================================================
 * 4. COLORI DELLO SFONDO (BACKGROUND) - PALETTE STANDARD (8 COLORI)
 * ====================================================================================
 * Cambiano il colore del blocco rettangolare dietro al carattere.
 */

#define TRM_BG_BLACK      "\x1b[40m"  /**< Sfondo Nero */
#define TRM_BG_RED        "\x1b[41m"  /**< Sfondo Rosso */
#define TRM_BG_GREEN      "\x1b[42m"  /**< Sfondo Verde */
#define TRM_BG_YELLOW     "\x1b[43m"  /**< Sfondo Giallo */
#define TRM_BG_BLUE       "\x1b[44m"  /**< Sfondo Blu */
#define TRM_BG_MAGENTA    "\x1b[45m"  /**< Sfondo Magenta / Viola */
#define TRM_BG_CYAN       "\x1b[46m"  /**< Sfondo Cyan / Azzurro */
#define TRM_BG_WHITE      "\x1b[47m"  /**< Sfondo Bianco */


/* ====================================================================================
 * 5. COLORI DELLO SFONDO (BACKGROUND) - ALTA INTENSITÀ / LUMINOSI (8 COLORI)
 * ====================================================================================
 */

#define TRM_BG_B_BLACK    "\x1b[100m" /**< Sfondo Grigio Scuro */
#define TRM_BG_B_RED      "\x1b[101m" /**< Sfondo Rosso Acceso */
#define TRM_BG_B_GREEN    "\x1b[102m" /**< Sfondo Verde Acceso */
#define TRM_BG_B_YELLOW   "\x1b[103m" /**< Sfondo Giallo Acceso */
#define TRM_BG_B_BLUE     "\x1b[104m" /**< Sfondo Blu Acceso */
#define TRM_BG_B_MAGENTA  "\x1b[105m" /**< Sfondo Magenta Acceso */
#define TRM_BG_B_CYAN     "\x1b[106m" /**< Sfondo Cyan Acceso */
#define TRM_BG_B_WHITE    "\x1b[107m" /**< Sfondo Bianco Luminoso */


/* ====================================================================================
 * 6. COMBINAZIONI PRONTE ALL'USO (LAYOUT AVANZATI)
 * ====================================================================================
 * Esempi di macro composte che uniscono stile, colore e sfondo in un'unica chiamata.
 */

#define CLI_BADGE_ERROR   "\x1b[1;97;41m" /**< Testo Grassetto Bianco su Sfondo Rosso (Per etichette [CRITICAL]) */
#define CLI_BADGE_SUCCESS "\x1b[1;30;102m"/**< Testo Grassetto Nero su Sfondo Verde Acceso (Per etichette [OK]) */
#define CLI_BADGE_WARN    "\x1b[1;30;103m"/**< Testo Grassetto Nero su Sfondo Giallo Acceso (Per etichette [WARN]) */
#define CLI_LOG_MUTED     "\x1b[2;90m"    /**< Testo Opaco + Grigio Scuro (Perfetto per log di debug o tracce secondarie) */


/* ====================================================================================
 * APPUNTI SUI COLORI AVANZATI (ESTESI):
 * ====================================================================================
 * Poiché i colori a 256 indici (8-bit) e i TrueColor (24-bit RGB) richiedono parametri variabili,
 * non sono definibili con macro statiche semplici. Ecco come usarli direttamente nel codice:
 * * >> TESTO A 256 COLORI:
 * printf("\x1b[38;5;[ID]mTuo Testo" TRM_RESET); -> Sostituisci [ID] con un numero da 0 a 255.
 * * >> TESTO TRUECOLOR (RGB):
 * printf("\x1b[38;2;[R];[G];[B]mTuo Testo" TRM_RESET); -> Sostituisci [R],[G],[B] con valori 0-255.
 * * >> SFONDO TRUECOLOR (RGB):
 * printf("\x1b[48;2;[R];[G];[B]mTuo Sfondo" TRM_RESET);
 * ====================================================================================
 */

#endif /* TERMINAL_COLORS_H */
